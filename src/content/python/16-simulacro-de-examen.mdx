        """Sustituye los espacios por el carácter (que no sea espacio) menos frecuente de toda la

        conteo = Counter("".join(serie).replace(" ", ""))

  solution={`
    import numpy as np
    import pandas as pd
    import matplotlib.pyplot as plt

    df = pd.read_csv("titanic.csv")

    df["Embarked"] = df["Embarked"].fillna(df["Embarked"].mode()[0])

    df["Familia"] = df["SibSp"] + df["Parch"] + 1

    df["Grupo"] = pd.cut(df["Familia"], bins=[0, 1, 4, np.inf], labels=["solo", "pequeña", "grande"])

    tasa_grupo = df.groupby("Grupo", observed=True)["Survived"].mean()

    mediana_puerto = df.groupby("Embarked")["Fare"].median()

    fig, axes = plt.subplots(1, 2, figsize=(10, 4))
    tasa_grupo.plot(kind="bar", ax=axes[0], title="Supervivencia por tamaño de familia")
    mediana_puerto.plot(kind="bar", ax=axes[1], title="Mediana del billete por puerto")
    fig.tight_layout()
    plt.show()

    print(tasa_grupo.round(3))
    print(mediana_puerto)
  `}
